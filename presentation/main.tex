\section{Background}
\subsection{The Interstellar Medium}
\begin{frame}{The Interstellar Medium}{The stuff between the stars}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{itemize}
        \item The interstellar medium (ISM) is the gas and dust between stars.
        \item Hot stars ionize hydrogen in the ISM, producing H$\alpha$ emission.
        \item ISM studies trace the structure of the Milky Way, providing information on Galactic evolution and star formation.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\linewidth}
      \centering

      \par\medskip
      {\footnotesize All-sky map of H$\alpha$ emission in the visible spectrum \cite{haffnerEarlyResultsWisconsin2010}.}
    \end{column}
  \end{columns}
\end{frame}

\subsection{Radio Recombination Lines}
\begin{frame}{Radio Recombination Lines}{The radio emission from ionized hydrogen}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{itemize}
        \item Hydrogen radio recombination lines (RRLs) are produced when ionized electrons recombine with protons and cascade through energy levels.
        \item Low energy transitions (for example, $n=100 \to 99$) produce emission in the radio spectrum.
        \item RRLs only occur at specific frequencies determined by the energy levels.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\linewidth}
      \centering
      {\usebeamercolor[fg]{structure}Bohr atomic model}
      \par\smallskip
      \includegraphics[width=0.8\linewidth]{rrl_model.png}
      \par\medskip
      {\footnotesize Simplified Bohr model of the hydrogen atom showing RRL emission from $N=100 \to 99$.}
    \end{column}
  \end{columns}
\end{frame}

\subsection{CHORD}
\begin{frame}{CHORD telescope}{The Canadian Hydrogen Observatory and Radio-transient Detector}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{itemize}
        \item  CHORD is a radio interferometer being constructed at the Dominion Radio Astro-physical Observatory (DRAO) in British Columbia, Canada.
        \item CHORD measures radio spectrum from 300 MHz to 1.5 GHz, which incumpasses 116 Hn$\alpha$ RRLs.
        \item CHORD will be able to observe declinations from $+20\deg$ to $+80\deg$, covering most of the northern sky.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\linewidth}
      \centering
      {\usebeamercolor[fg]{structure}CHORD under construction}
      \par\smallskip
      \includegraphics[width=0.8\linewidth]{CHORD.jpeg}
      \par\medskip
      {\footnotesize The CHORD array, once complete, will consist of 512 6-meter dishes \cite{SPIEpaper}}
    \end{column}
  \end{columns}
\end{frame}
